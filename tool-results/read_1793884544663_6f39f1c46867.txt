    19→/**
    20→ * Studio laptop render — ported from the Gemini-authored Bitify laptop
    21→ * render, adapted for the Bitify hero:
    22→ *
    23→ *  - The screen now carries the REAL Bitify desktop screenshot
    24→ *    (`desktop-hero.webp`, 1403 x 930, ratio 1.5086). The physical
    25→ *    screen aperture and the lid height were re-derived so the
    26→ *    screenshot maps onto the display with ZERO crop, stretch or
    27→ *    letterboxing, while keeping realistic slim bezels.
    28→ *  - The side ports are drawn on the wall that actually faces the
    29→ *    camera for negative yaw (the original drew them on the far
    30→ *    wall, which made them float over the keyboard).
    31→ *  - The camera viewport is recentered so the whole laptop sits
    32→ *    dead-center inside the 1440 x 960 stage (predictable CSS sizing).
    33→ *
    34→ * The laptop is built from a real pinhole-camera projection: every
    35→ * chassis edge, wall, hinge and shadow is a 3D point projected to 2D,
    36→ * and the screen + keyboard deck are flat surfaces mapped on with
    37→ * solved homography matrix3d transforms. The default pose faces the
    38→ * viewer with a slight left bias — perfect for a right-hand hero
    39→ * placement looking back at the headline. A second, dead-on FRONT
    40→ * pose (via the `camera` prop) powers the desktop showcase section,
    41→ * and `children` swaps the screen content (live video / tab views).
    42→ */
    43→
    44→/* Real screenshot pixel size — the screen div is exactly this big so
    45→ * the image maps 1:1 onto the projected screen quad. */
    46→export const SCREEN_PIXEL_WIDTH = 1403;
    47→export const SCREEN_PIXEL_HEIGHT = 930;
    48→const SCREEN_SRC = "/assets/desktop/desktop-hero.webp";
    49→
    50→export const STAGE_WIDTH = 1440;
    51→export const STAGE_HEIGHT = 960;
    52→
    53→const KEY_LIGHT = 1.0;
    54→const CRIMSON_RIM = 1.0;
    55→const SCREEN_BRIGHTNESS = 1.02;
    56→const SHADOW_INTENSITY = 1.0;
    57→
    58→/* Fixed hero pose — three-quarter view, deck front angled toward the
    59→ * lower-left, screen normal facing slightly left (toward the copy).
    60→ * Hand-tuned so the projected laptop (incl. contact shadow) is
    61→ * horizontally and vertically centered in the 1440x960 stage. */
    62→export const HERO_CAMERA: CameraConfig = {
    63→  yawDeg: -19.5,
    64→  pitchDeg: 17.5,
    65→  lidAngleDeg: 102.5,
    66→  distance: 1060,
    67→  focalLength: 2280,
    68→  target: { x: -8, y: 60, z: 98 },
    69→  viewportCenter: { x: 726.35, y: 517.05 },
    70→};
    71→
    72→/* Front-facing product pose — dead-on and perfectly symmetric (yaw 0),
    73→ * pitch 16° still reveals the keyboard deck below the screen. Derived
    74→ * by projecting the full chassis + contact-shadow bounding box and
    75→ * solving the focal length + viewport centre that centres it in the
    76→ * 1440x960 stage: the laptop fills ~79% of the width and ~92% of the
    77→ * height — a touch larger than the hero crop, with just enough margin
    78→ * for the contact-shadow blur to breathe. */
    79→export const FRONT_CAMERA: CameraConfig = {
    80→  yawDeg: 0,
    81→  pitchDeg: 16,
    82→  lidAngleDeg: 102.5,
    83→  distance: 1060,
    84→  focalLength: 2825.9,
    85→  target: { x: 0, y: 60, z: 98 },
    86→  viewportCenter: { x: 720.0, y: 562.83 },
    87→};
    88→
    89→export const StudioLaptopRender: React.FC<{
    90→  /** Camera pose — defaults to the hero three-quarter view. */
    91→  camera?: CameraConfig;
    92→  /**
    93→   * Optional screen content, rendered inside the projected display
    94→   * aperture (a 1403x930 surface homography-mapped onto the screen
    95→   * quad) in place of the default hero screenshot. Fill it with
    96→   * absolutely-positioned media (video / next-image) — it inherits
    97→   * the laptop's perspective exactly.
    98→   */
    99→  children?: ReactNode;
   100→}> = ({ camera = HERO_CAMERA, children }) => {
   101→  const cam = camera;
   102→  /* Unique SVG def ids per instance — the page mounts this laptop
   103→   * twice (hero + desktop showcase) and duplicate filter/gradient
   104→   * ids would collide across the two SVGs. */
   105→  const uid = useId().replace(/[^a-zA-Z0-9]/g, "");
   106→  const svgId = (name: string) => `sl-${uid}-${name}`;
   107→
   108→  // Physical dimensions in millimeter-scale 3D space
   109→  const halfW = 178; // 356mm total width
   110→  const deckDepth = 246; // 246mm deck depth
   111→  // Lid tall enough for a 1403x930 (1.5086) screen with slim bezels:
   112→  // screen 344.8 x 228.56 -> bottom bezel 8.2mm, top bezel 5.84mm
   113→  const lidHeight = 242.6;
   114→  const lidThickness = 5.0; // Slim 5.0mm anodized aluminum lid
   115→  const baseWallBottomY = -10.8; // Vertical side wall bottom
   116→  const basePanBottomY = -13.8; // Tapered lower chassis pan
   117→
   118→  // Convert display lid local coordinates (u, v, w) to 3D world space
   119→  // u: [-halfW..+halfW], v: [0..lidHeight] along lid face, w: [-lidThickness..0] depth into back shell
   120→  const lidPointToWorld = useMemo(() => {
   121→    const tiltBackRad = ((cam.lidAngleDeg - 90) * Math.PI) / 180;
   122→    const cosT = Math.cos(tiltBackRad);
   123→    const sinT = Math.sin(tiltBackRad);
   124→    const hingeY = 1.2;
   125→    const hingeZ = 7.5;
   126→
   127→    return (u: number, v: number, w = 0): Vec3 => ({
   128→      x: u,
   129→      y: hingeY + v * cosT - w * sinT,
   130→      z: hingeZ - v * sinT + w * cosT,
   131→    });
   132→  }, [cam]);
   133→
   134→  // 1. Compute 4 Projected Corners of the Active Display Screen Aperture
   135→  // Physical screen aperture: u in [-172.4, +172.4] (344.8mm), v in [8.2, 236.76]
   136→  // (228.56mm) -> exact 1403:930 = the real screenshot's native ratio
   137→  const screenQuad = useMemo((): [Point2D, Point2D, Point2D, Point2D] => {
   138→    const uMin = -172.4;
   139→    const uMax = 172.4;
   140→    const vBottom = 8.2;
   141→    const vTop = 236.76;
   142→
   143→    const tl = projectPoint3D(lidPointToWorld(uMin, vTop, 0), cam);
   144→    const tr = projectPoint3D(lidPointToWorld(uMax, vTop, 0), cam);
   145→    const br = projectPoint3D(lidPointToWorld(uMax, vBottom, 0), cam);
   146→    const bl = projectPoint3D(lidPointToWorld(uMin, vBottom, 0), cam);
   147→    return [tl, tr, br, bl];
   148→  }, [cam, lidPointToWorld]);
   149→
   150→  const screenMatrix3d = useMemo(
   151→    () =>
   152→      computeHomographyMatrix3d(
   153→        SCREEN_PIXEL_WIDTH,
   154→        SCREEN_PIXEL_HEIGHT,
   155→        screenQuad
   156→      ),
   157→    [screenQuad]
   158→  );
   159→
   160→  // 2. Compute 4 Projected Corners of the Top Keyboard Deck Plane (Y = 0)
   161→  const deckQuad = useMemo((): [Point2D, Point2D, Point2D, Point2D] => {
   162→    const tl = projectPoint3D({ x: -halfW, y: 0, z: 0 }, cam);
   163→    const tr = projectPoint3D({ x: halfW, y: 0, z: 0 }, cam);
   164→    const br = projectPoint3D({ x: halfW, y: 0, z: deckDepth }, cam);
   165→    const bl = projectPoint3D({ x: -halfW, y: 0, z: deckDepth }, cam);
   166→    return [tl, tr, br, bl];
   167→  }, [cam, halfW, deckDepth]);
   168→
   169→  const deckMatrix3d = useMemo(
   170→    () => computeHomographyMatrix3d(DECK_PIXEL_WIDTH, DECK_PIXEL_HEIGHT, deckQuad),
   171→    [deckQuad]
   172→  );
   173→
   174→  // 3. Sample 3D Rounded Contours for Lid, Screen Bezel, Deck, Lower Chassis, and Shadows
   175→  const geometryPaths = useMemo(() => {
   176→    // Lid 2D local rounded contour (note: in 2D sampling yMin is bottom v=0, yMax is top v=lidHeight)
   177→    const lidContourLocal = sampleRoundedRect2D(
   178→      -halfW,
   179→      halfW,
   180→      0,
   181→      lidHeight,
   182→      [5, 5, 11.5, 11.5],
   183→      12
   184→    );
   185→
   186→    const lidFrontPts = lidContourLocal.map((p) =>
   187→      projectPoint3D(lidPointToWorld(p.x, p.y, 0), cam)
   188→    );
   189→    const lidBackPts = lidContourLocal.map((p) =>
   190→      projectPoint3D(lidPointToWorld(p.x, p.y, -lidThickness), cam)
   191→    );
   192→
   193→    // Lid inner bezel step (1.5mm inset)
   194→    const lidInnerPts = sampleRoundedRect2D(
   195→      -halfW + 1.6,
   196→      halfW - 1.6,
   197→      1.4,
   198→      lidHeight - 1.6,
   199→      [4, 4, 10, 10],
   200→      10
   201→    ).map((p) => projectPoint3D(lidPointToWorld(p.x, p.y, 0), cam));
   202→
   203→    // Build 3D extruded rim quads between lidBackPts and lidFrontPts
   204→    const lidRimSegments: {
   205→      d: string;
   206→      color: string;
   207→      isTopKeyLight: boolean;
   208→      isCrimsonRim: boolean;
   209→    }[] = [];
   210→
   211→    for (let i = 0; i < lidFrontPts.length; i++) {
   212→      const next = (i + 1) % lidFrontPts.length;
   213→      const pL1 = lidContourLocal[i];
   214→      const pL2 = lidContourLocal[next];
   215→      const midU = (pL1.x + pL2.x) * 0.5;
   216→      const midV = (pL1.y + pL2.y) * 0.5;
   217→
   218→      const isTop = midV > lidHeight - 14;
   219→      const isRightSide = midU > halfW - 14;
   220→      const isLeftSide = midU < -halfW + 14;
   221→
   222→      let fill = "#16161B";
   223→      if (isTop) {
   224→        fill = "#383436"; // Warm-white key light grazing upper edge
   225→      } else if (isRightSide || isLeftSide) {
   226→        fill = "#2A1720"; // Subtle crimson rim light along outer edges
   227→      }
   228→
   229→      const segPath = pointsToSvgPath(
   230→        [lidBackPts[i], lidBackPts[next], lidFrontPts[next], lidFrontPts[i]],
   231→        true
   232→      );
   233→      lidRimSegments.push({
   234→        d: segPath,
   235→        color: fill,
   236→        isTopKeyLight: isTop,
   237→        isCrimsonRim: isRightSide || isLeftSide,
   238→      });
   239→    }
   240→
   241→    // Top camera notch/sensor coordinates on upper bezel
   242→    const camCenter = projectPoint3D(
   243→      lidPointToWorld(0, lidHeight - 3.8, 0),
   244→      cam
   245→    );
   246→    const camSensorLeft = projectPoint3D(
   247→      lidPointToWorld(-4.5, lidHeight - 3.8, 0),
   248→      cam
   249→    );
   250→
   251→    // Cylindrical hinge barrel between (X in [-132, +132])
   252→    const hingeTopLeft = projectPoint3D({ x: -134, y: 4.2, z: 6.5 }, cam);
   253→    const hingeTopRight = projectPoint3D({ x: 134, y: 4.2, z: 6.5 }, cam);
   254→    const hingeBotRight = projectPoint3D({ x: 134, y: -1.2, z: 11.5 }, cam);
   255→    const hingeBotLeft = projectPoint3D({ x: -134, y: -1.2, z: 11.5 }, cam);
   256→    const hingePath = pointsToSvgPath([
   257→      hingeTopLeft,
   258→      hingeTopRight,
   259→      hingeBotRight,
   260→      hingeBotLeft,
   261→    ]);
   262→
   263→    // Deck 2D local contour: X in [-halfW..+halfW], Z in [0..deckDepth]
   264→    const deckContourLocal = sampleRoundedRect2D(
   265→      -halfW,
   266→      halfW,
   267→      0,
   268→      deckDepth,
   269→      [11, 11, 13.5, 13.5],
   270→      14
   271→    );
   272→
   273→    const deckTopPts = deckContourLocal.map((p) =>
   274→      projectPoint3D({ x: p.x, y: 0, z: p.y }, cam)
   275→    );
   276→    const deckMidPts = deckContourLocal.map((p) =>
   277→      projectPoint3D({ x: p.x, y: baseWallBottomY, z: p.y }, cam)
   278→    );
   279→
   280→    // Inset tapered lower pan contour
   281→    const panContourLocal = sampleRoundedRect2D(
   282→      -halfW + 5.5,
   283→      halfW - 5.5,
   284→      5.5,
   285→      deckDepth - 5.5,
   286→      [10, 10, 12, 12],
   287→      12
   288→    );
   289→    const deckPanPts = panContourLocal.map((p) =>
   290→      projectPoint3D({ x: p.x, y: basePanBottomY, z: p.y }, cam)
   291→    );
   292→
   293→    // Build 3D vertical wall segments around the unibody deck perimeter
   294→    const deckWallSegments: { d: string; fill: string }[] = [];
   295→    for (let i = 0; i < deckTopPts.length; i++) {
   296→      const next = (i + 1) % deckTopPts.length;
   297→      const c1 = deckContourLocal[i];
   298→      const c2 = deckContourLocal[next];
   299→
   300→      // Outward 2D normal in (X, Z) plane
   301→      const dx = c2.x - c1.x;
   302→      const dz = c2.y - c1.y;
   303→      const len = Math.hypot(dx, dz) || 1;
   304→      const nx = dz / len;
   305→      const nz = -dx / len;
   306→
   307→      // Shade based on normal orientation (front lip vs left/right side walls vs corner curve)
   308→      let r = 20;
   309→      let g = 20;
   310→      let b = 25;
   311→
   312→      // Front-facing lip receives soft warm studio fill
   313→      if (nz > 0.4) {
   314→        const frontFactor = (nz - 0.4) / 0.6;
   315→        r += Math.round(frontFactor * 14 * KEY_LIGHT);
   316→        g += Math.round(frontFactor * 13 * KEY_LIGHT);
   317→        b += Math.round(frontFactor * 15 * KEY_LIGHT);
   318→      }
   319→
   320→      // Corner specular catchlight where side wall curves into front lip
   321→      const cornerHighlight = Math.max(0, 1 - Math.abs(Math.abs(nx) - 0.7) * 3.2);
   322→      if (nz > 0) {
   323→        r += Math.round(cornerHighlight * 28 * KEY_LIGHT);
   324→        g += Math.round(cornerHighlight * 25 * KEY_LIGHT);
   325→        b += Math.round(cornerHighlight * 27 * KEY_LIGHT);
   326→      }
   327→
   328→      // Crimson rim light along outer side edges
   329→      if (Math.abs(nx) > 0.55) {
   330→        r += Math.round(24 * CRIMSON_RIM);
   331→        g += Math.round(4 * CRIMSON_RIM);
   332→        b += Math.round(10 * CRIMSON_RIM);
   333→      }
   334→
   335→      const segPath = pointsToSvgPath([
   336→        deckTopPts[i],
   337→        deckTopPts[next],
   338→        deckMidPts[next],
   339→        deckMidPts[i],
   340→      ]);
   341→      deckWallSegments.push({
   342→        d: segPath,
   343→        fill: `rgb(${Math.min(255, r)}, ${Math.min(255, g)}, ${Math.min(255, b)})`,
   344→      });
   345→    }
   346→
   347→    // Build tapered lower bevel segments between deckMidPts and deckPanPts
   348→    const lowerChamferPoly = pointsToSvgPath(deckMidPts);
   349→    const lowerPanPoly = pointsToSvgPath(deckPanPts);
   350→
   351→    // Front lip ergonomic CNC thumb notch (X in [-27, +27], Z = deckDepth, Y in [-4.4, 0])
   352→    const notchPts = [
   353→      projectPoint3D({ x: -27, y: 0, z: deckDepth }, cam),
   354→      projectPoint3D({ x: 27, y: 0, z: deckDepth }, cam),
   355→      projectPoint3D({ x: 24.5, y: -4.4, z: deckDepth }, cam),
   356→      projectPoint3D({ x: -24.5, y: -4.4, z: deckDepth }, cam),
   357→    ];
   358→    const notchBottomRim = [
   359→      projectPoint3D({ x: -24.5, y: -4.4, z: deckDepth }, cam),
   360→      projectPoint3D({ x: 24.5, y: -4.4, z: deckDepth }, cam),
   361→    ];
   362→
   363→    // Side ports on the wall that FACES the camera:
   364→    // with negative yaw the camera sits at +X, so the +halfW wall is
   365→    // visible. Dead-on (|yaw| < 3°) both side walls are edge-on — the
   366→    // port cutouts would collapse to zero-width slivers, so they are
   367→    // omitted entirely for the front pose.
   368→    const sideX = cam.yawDeg <= 0 ? halfW : -halfW;
   369→    const makeSidePort = (zStart: number, zEnd: number, yTop: number, yBot: number) =>
   370→      pointsToSvgPath([
   371→        projectPoint3D({ x: sideX, y: yTop, z: zStart }, cam),
   372→        projectPoint3D({ x: sideX, y: yTop, z: zEnd }, cam),
   373→        projectPoint3D({ x: sideX, y: yBot, z: zEnd }, cam),
   374→        projectPoint3D({ x: sideX, y: yBot, z: zStart }, cam),
   375→      ]);
   376→
   377→    const sidePorts =
   378→      Math.abs(cam.yawDeg) < 3
   379→        ? []
   380→        : [
   381→            // MagSafe 3 Port
   382→            { d: makeSidePort(28, 46, -2.8, -7.6), isMagSafe: true },
   383→            // Thunderbolt / USB-C Port 1
   384→            { d: makeSidePort(58, 71, -3.4, -7.2), isMagSafe: false },
   385→            // Thunderbolt / USB-C Port 2
   386→            { d: makeSidePort(82, 95, -3.4, -7.2), isMagSafe: false },
   387→            // 3.5mm Studio Audio Jack
   388→            { d: makeSidePort(110, 116, -3.8, -7.0), isMagSafe: false },
   389→          ];
   390→
   391→    // Four Rubber Feet underneath the chassis corners + their tight contact shadows
   392→    const footConfigs = [
   393→      { x: -halfW + 26, z: 26, r: 12 },
   394→      { x: halfW - 26, z: 26, r: 12 },
   395→      { x: -halfW + 26, z: deckDepth - 26, r: 12 },
   396→      { x: halfW - 26, z: deckDepth - 26, r: 12 },
   397→    ];
   398→
   399→    const rubberFeet = footConfigs.map((fc) => {
   400→      const topLoop: Point2D[] = [];
   401→      const botLoop: Point2D[] = [];
   402→      const shadowLoop: Point2D[] = [];
   403→      const steps = 16;
   404→      for (let i = 0; i < steps; i++) {
   405→        const a = (i / steps) * Math.PI * 2;
   406→        const px = fc.x + Math.cos(a) * fc.r;
   407→        const pz = fc.z + Math.sin(a) * fc.r;
   408→        topLoop.push(projectPoint3D({ x: px, y: basePanBottomY + 0.5, z: pz }, cam));
   409→        botLoop.push(projectPoint3D({ x: px, y: basePanBottomY - 2.2, z: pz }, cam));
   410→        shadowLoop.push(
   411→          projectPoint3D(
   412→            {
   413→              x: fc.x + Math.cos(a) * (fc.r * 1.45),
   414→              y: basePanBottomY - 2.5,
   415→              z: fc.z + Math.sin(a) * (fc.r * 1.45),
   416→            },
   417→            cam
   418→          )
   419→        );
   420→      }
   421→      return {
   422→        footPath: pointsToSvgPath(botLoop),
   423→        shadowPath: pointsToSvgPath(shadowLoop),
   424→      };
   425→    });
   426→
   427→    // Studio Contact Shadows on the ground plane (Y = basePanBottomY - 2.6)
   428→    // Kept tight and natural directly beneath the laptop chassis
   429→    const shadowOuterPts = sampleRoundedRect2D(
   430→      -halfW - 10,
   431→      halfW + 10,
   432→      -8,
   433→      deckDepth + 10,
   434→      [22, 22, 26, 26],
   435→      12
   436→    ).map((p) =>
   437→      projectPoint3D({ x: p.x, y: basePanBottomY - 2.8, z: p.y }, cam)
   438→    );
   439→
   440→    const shadowCorePts = sampleRoundedRect2D(
   441→      -halfW + 4,
   442→      halfW - 4,
   443→      4,
   444→      deckDepth - 2,
   445→      [14, 14, 16, 16],
   446→      12
   447→    ).map((p) =>
   448→      projectPoint3D({ x: p.x, y: basePanBottomY - 2.4, z: p.y }, cam)
   449→    );
   450→
   451→    // Upper display lid warm-white key light highlight path
   452→    const topKeyLightPts: Point2D[] = [];
   453→    for (let u = -halfW + 10; u <= halfW - 10; u += 12) {
   454→      topKeyLightPts.push(
   455→        projectPoint3D(lidPointToWorld(u, lidHeight - 0.4, -0.5), cam)
   456→      );
   457→    }
   458→
   459→    // Outer display lid crimson rim highlight paths
   460→    const rightCrimsonRimPts: Point2D[] = [];
   461→    for (let v = 10; v <= lidHeight - 10; v += 12) {
   462→      rightCrimsonRimPts.push(
   463→        projectPoint3D(lidPointToWorld(halfW - 0.3, v, -0.8), cam)
   464→      );
   465→    }
   466→
   467→    const leftCrimsonRimPts: Point2D[] = [];
   468→    for (let v = 10; v <= lidHeight - 10; v += 12) {
   469→      leftCrimsonRimPts.push(
   470→        projectPoint3D(lidPointToWorld(-halfW + 0.3, v, -0.8), cam)
   471→      );
   472→    }
   473→
   474→    return {
   475→      lidFrontPath: pointsToSvgPath(lidFrontPts),
   476→      lidBackPath: pointsToSvgPath(lidBackPts),
   477→      lidInnerPath: pointsToSvgPath(lidInnerPts),
   478→      lidRimSegments,
   479→      camCenter,
   480→      camSensorLeft,
   481→      hingePath,
   482→      deckTopPath: pointsToSvgPath(deckTopPts),
   483→      deckMidPath: lowerChamferPoly,
   484→      deckPanPath: lowerPanPoly,
   485→      deckWallSegments,
   486→      notchPath: pointsToSvgPath(notchPts),
   487→      notchBottomRim,
   488→      sidePorts,
   489→      rubberFeet,
   490→      shadowOuterPath: pointsToSvgPath(shadowOuterPts),
   491→      shadowCorePath: pointsToSvgPath(shadowCorePts),
   492→      topKeyLightPath: pointsToSvgPath(topKeyLightPts, false),
   493→      rightCrimsonRimPath: pointsToSvgPath(rightCrimsonRimPts, false),
   494→      leftCrimsonRimPath: pointsToSvgPath(leftCrimsonRimPts, false),
   495→    };
   496→  }, [
   497→    cam,
