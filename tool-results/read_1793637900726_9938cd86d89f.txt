     1→"use client";
     2→
     3→import React, { useMemo } from "react";
     4→import {
     5→  type CameraConfig,
     6→  type Point2D,
     7→  type Vec3,
     8→  computeHomographyMatrix3d,
     9→  pointsToSvgPath,
    10→  projectPoint3D,
    11→  sampleRoundedRect2D,
    12→} from "./homography";
    13→import {
    14→  DECK_PIXEL_HEIGHT,
    15→  DECK_PIXEL_WIDTH,
    16→  LaptopKeyboardDeck,
    17→} from "./keyboard-deck";
    18→
    19→/**
    20→ * Studio laptop render — ported from the Gemini-authored Bitify laptop
    21→ * render, adapted for the Bitify hero:
    22→ *
    23→ *  - The screen now carries the REAL Bitify desktop screenshot
    24→ *    (`desktop-hero.webp`, 1403 x 930, ratio 1.5086). The physical
    25→ *    screen aperture and the lid height were re-derived so the
    26→ *    screenshot maps onto the display with ZERO crop, stretch or
    27→ *    letterboxing, while keeping realistic slim bezels.
    28→ *  - The side ports are drawn on the wall that actually faces the
    29→ *    camera for negative yaw (the original drew them on the far
    30→ *    wall, which made them float over the keyboard).
    31→ *  - The camera viewport is recentered so the whole laptop sits
    32→ *    dead-center inside the 1440 x 960 stage (predictable CSS sizing).
    33→ *
    34→ * The laptop is built from a real pinhole-camera projection: every
    35→ * chassis edge, wall, hinge and shadow is a 3D point projected to 2D,
    36→ * and the screen + keyboard deck are flat surfaces mapped on with
    37→ * solved homography matrix3d transforms. The default pose faces the
    38→ * viewer with a slight left bias — perfect for a right-hand hero
    39→ * placement looking back at the headline.
    40→ */
    41→
    42→/* Real screenshot pixel size — the screen div is exactly this big so
    43→ * the image maps 1:1 onto the projected screen quad. */
    44→export const SCREEN_PIXEL_WIDTH = 1403;
    45→export const SCREEN_PIXEL_HEIGHT = 930;
    46→const SCREEN_SRC = "/assets/desktop/desktop-hero.webp";
    47→
    48→export const STAGE_WIDTH = 1440;
    49→export const STAGE_HEIGHT = 960;
    50→
    51→/* Fixed hero pose — three-quarter view, deck front angled toward the
    52→ * lower-left, screen normal facing slightly left (toward the copy). */
    53→const YAW_DEG = -19.5;
    54→const PITCH_DEG = 17.5;
    55→const LID_ANGLE_DEG = 102.5;
    56→const KEY_LIGHT = 1.0;
    57→const CRIMSON_RIM = 1.0;
    58→const SCREEN_BRIGHTNESS = 1.02;
    59→const SHADOW_INTENSITY = 1.0;
    60→
    61→export const StudioLaptopRender: React.FC = () => {
    62→  const cam: CameraConfig = useMemo(
    63→    () => ({
    64→      yawDeg: YAW_DEG,
    65→      pitchDeg: PITCH_DEG,
    66→      lidAngleDeg: LID_ANGLE_DEG,
    67→      distance: 1060,
    68→      focalLength: 2280,
    69→      target: { x: -8, y: 60, z: 98 },
    70→      /* recentred so the projected laptop (incl. contact shadow) is
    71→       * horizontally and vertically centered in the 1440x960 stage */
    72→      viewportCenter: { x: 726.35, y: 517.05 },
    73→    }),
    74→    []
    75→  );
    76→
    77→  // Physical dimensions in millimeter-scale 3D space
    78→  const halfW = 178; // 356mm total width
    79→  const deckDepth = 246; // 246mm deck depth
    80→  // Lid tall enough for a 1403x930 (1.5086) screen with slim bezels:
    81→  // screen 344.8 x 228.56 -> bottom bezel 8.2mm, top bezel 5.84mm
    82→  const lidHeight = 242.6;
    83→  const lidThickness = 5.0; // Slim 5.0mm anodized aluminum lid
    84→  const baseWallBottomY = -10.8; // Vertical side wall bottom
    85→  const basePanBottomY = -13.8; // Tapered lower chassis pan
    86→
    87→  // Convert display lid local coordinates (u, v, w) to 3D world space
    88→  // u: [-halfW..+halfW], v: [0..lidHeight] along lid face, w: [-lidThickness..0] depth into back shell
    89→  const lidPointToWorld = useMemo(() => {
    90→    const tiltBackRad = ((LID_ANGLE_DEG - 90) * Math.PI) / 180;
    91→    const cosT = Math.cos(tiltBackRad);
    92→    const sinT = Math.sin(tiltBackRad);
    93→    const hingeY = 1.2;
    94→    const hingeZ = 7.5;
    95→
    96→    return (u: number, v: number, w = 0): Vec3 => ({
    97→      x: u,
    98→      y: hingeY + v * cosT - w * sinT,
    99→      z: hingeZ - v * sinT + w * cosT,
   100→    });
   101→  }, []);
   102→
   103→  // 1. Compute 4 Projected Corners of the Active Display Screen Aperture
   104→  // Physical screen aperture: u in [-172.4, +172.4] (344.8mm), v in [8.2, 236.76]
   105→  // (228.56mm) -> exact 1403:930 = the real screenshot's native ratio
   106→  const screenQuad = useMemo((): [Point2D, Point2D, Point2D, Point2D] => {
   107→    const uMin = -172.4;
   108→    const uMax = 172.4;
   109→    const vBottom = 8.2;
   110→    const vTop = 236.76;
   111→
   112→    const tl = projectPoint3D(lidPointToWorld(uMin, vTop, 0), cam);
   113→    const tr = projectPoint3D(lidPointToWorld(uMax, vTop, 0), cam);
   114→    const br = projectPoint3D(lidPointToWorld(uMax, vBottom, 0), cam);
   115→    const bl = projectPoint3D(lidPointToWorld(uMin, vBottom, 0), cam);
   116→    return [tl, tr, br, bl];
   117→  }, [cam, lidPointToWorld]);
   118→
   119→  const screenMatrix3d = useMemo(
   120→    () =>
   121→      computeHomographyMatrix3d(
   122→        SCREEN_PIXEL_WIDTH,
   123→        SCREEN_PIXEL_HEIGHT,
   124→        screenQuad
   125→      ),
   126→    [screenQuad]
   127→  );
   128→
   129→  // 2. Compute 4 Projected Corners of the Top Keyboard Deck Plane (Y = 0)
   130→  const deckQuad = useMemo((): [Point2D, Point2D, Point2D, Point2D] => {
   131→    const tl = projectPoint3D({ x: -halfW, y: 0, z: 0 }, cam);
   132→    const tr = projectPoint3D({ x: halfW, y: 0, z: 0 }, cam);
   133→    const br = projectPoint3D({ x: halfW, y: 0, z: deckDepth }, cam);
   134→    const bl = projectPoint3D({ x: -halfW, y: 0, z: deckDepth }, cam);
   135→    return [tl, tr, br, bl];
   136→  }, [cam, halfW, deckDepth]);
   137→
   138→  const deckMatrix3d = useMemo(
   139→    () => computeHomographyMatrix3d(DECK_PIXEL_WIDTH, DECK_PIXEL_HEIGHT, deckQuad),
   140→    [deckQuad]
   141→  );
   142→
   143→  // 3. Sample 3D Rounded Contours for Lid, Screen Bezel, Deck, Lower Chassis, and Shadows
   144→  const geometryPaths = useMemo(() => {
   145→    // Lid 2D local rounded contour (note: in 2D sampling yMin is bottom v=0, yMax is top v=lidHeight)
   146→    const lidContourLocal = sampleRoundedRect2D(
   147→      -halfW,
   148→      halfW,
   149→      0,
   150→      lidHeight,
   151→      [5, 5, 11.5, 11.5],
   152→      12
   153→    );
   154→
   155→    const lidFrontPts = lidContourLocal.map((p) =>
   156→      projectPoint3D(lidPointToWorld(p.x, p.y, 0), cam)
   157→    );
   158→    const lidBackPts = lidContourLocal.map((p) =>
   159→      projectPoint3D(lidPointToWorld(p.x, p.y, -lidThickness), cam)
   160→    );
   161→
   162→    // Lid inner bezel step (1.5mm inset)
   163→    const lidInnerPts = sampleRoundedRect2D(
   164→      -halfW + 1.6,
   165→      halfW - 1.6,
   166→      1.4,
   167→      lidHeight - 1.6,
   168→      [4, 4, 10, 10],
   169→      10
   170→    ).map((p) => projectPoint3D(lidPointToWorld(p.x, p.y, 0), cam));
   171→
   172→    // Build 3D extruded rim quads between lidBackPts and lidFrontPts
   173→    const lidRimSegments: {
   174→      d: string;
   175→      color: string;
   176→      isTopKeyLight: boolean;
   177→      isCrimsonRim: boolean;
   178→    }[] = [];
   179→
   180→    for (let i = 0; i < lidFrontPts.length; i++) {
   181→      const next = (i + 1) % lidFrontPts.length;
   182→      const pL1 = lidContourLocal[i];
   183→      const pL2 = lidContourLocal[next];
   184→      const midU = (pL1.x + pL2.x) * 0.5;
   185→      const midV = (pL1.y + pL2.y) * 0.5;
   186→
   187→      const isTop = midV > lidHeight - 14;
   188→      const isRightSide = midU > halfW - 14;
   189→      const isLeftSide = midU < -halfW + 14;
   190→
   191→      let fill = "#16161B";
   192→      if (isTop) {
   193→        fill = "#383436"; // Warm-white key light grazing upper edge
   194→      } else if (isRightSide || isLeftSide) {
   195→        fill = "#2A1720"; // Subtle crimson rim light along outer edges
   196→      }
   197→
   198→      const segPath = pointsToSvgPath(
   199→        [lidBackPts[i], lidBackPts[next], lidFrontPts[next], lidFrontPts[i]],
   200→        true
   201→      );
   202→      lidRimSegments.push({
   203→        d: segPath,
   204→        color: fill,
   205→        isTopKeyLight: isTop,
   206→        isCrimsonRim: isRightSide || isLeftSide,
   207→      });
   208→    }
   209→
   210→    // Top camera notch/sensor coordinates on upper bezel
   211→    const camCenter = projectPoint3D(
   212→      lidPointToWorld(0, lidHeight - 3.8, 0),
   213→      cam
   214→    );
   215→    const camSensorLeft = projectPoint3D(
   216→      lidPointToWorld(-4.5, lidHeight - 3.8, 0),
   217→      cam
   218→    );
   219→
   220→    // Cylindrical hinge barrel between (X in [-132, +132])
   221→    const hingeTopLeft = projectPoint3D({ x: -134, y: 4.2, z: 6.5 }, cam);
   222→    const hingeTopRight = projectPoint3D({ x: 134, y: 4.2, z: 6.5 }, cam);
   223→    const hingeBotRight = projectPoint3D({ x: 134, y: -1.2, z: 11.5 }, cam);
   224→    const hingeBotLeft = projectPoint3D({ x: -134, y: -1.2, z: 11.5 }, cam);
   225→    const hingePath = pointsToSvgPath([
   226→      hingeTopLeft,
   227→      hingeTopRight,
   228→      hingeBotRight,
   229→      hingeBotLeft,
   230→    ]);
   231→
   232→    // Deck 2D local contour: X in [-halfW..+halfW], Z in [0..deckDepth]
   233→    const deckContourLocal = sampleRoundedRect2D(
   234→      -halfW,
   235→      halfW,
   236→      0,
   237→      deckDepth,
   238→      [11, 11, 13.5, 13.5],
   239→      14
   240→    );
   241→
   242→    const deckTopPts = deckContourLocal.map((p) =>
   243→      projectPoint3D({ x: p.x, y: 0, z: p.y }, cam)
   244→    );
   245→    const deckMidPts = deckContourLocal.map((p) =>
   246→      projectPoint3D({ x: p.x, y: baseWallBottomY, z: p.y }, cam)
   247→    );
   248→
   249→    // Inset tapered lower pan contour
   250→    const panContourLocal = sampleRoundedRect2D(
   251→      -halfW + 5.5,
   252→      halfW - 5.5,
   253→      5.5,
   254→      deckDepth - 5.5,
   255→      [10, 10, 12, 12],
   256→      12
   257→    );
   258→    const deckPanPts = panContourLocal.map((p) =>
   259→      projectPoint3D({ x: p.x, y: basePanBottomY, z: p.y }, cam)
   260→    );
   261→
   262→    // Build 3D vertical wall segments around the unibody deck perimeter
   263→    const deckWallSegments: { d: string; fill: string }[] = [];
   264→    for (let i = 0; i < deckTopPts.length; i++) {
   265→      const next = (i + 1) % deckTopPts.length;
   266→      const c1 = deckContourLocal[i];
   267→      const c2 = deckContourLocal[next];
   268→
   269→      // Outward 2D normal in (X, Z) plane
   270→      const dx = c2.x - c1.x;
   271→      const dz = c2.y - c1.y;
   272→      const len = Math.hypot(dx, dz) || 1;
   273→      const nx = dz / len;
   274→      const nz = -dx / len;
   275→
   276→      // Shade based on normal orientation (front lip vs left/right side walls vs corner curve)
   277→      let r = 20;
   278→      let g = 20;
   279→      let b = 25;
   280→
   281→      // Front-facing lip receives soft warm studio fill
   282→      if (nz > 0.4) {
   283→        const frontFactor = (nz - 0.4) / 0.6;
   284→        r += Math.round(frontFactor * 14 * KEY_LIGHT);
   285→        g += Math.round(frontFactor * 13 * KEY_LIGHT);
   286→        b += Math.round(frontFactor * 15 * KEY_LIGHT);
   287→      }
   288→
   289→      // Corner specular catchlight where side wall curves into front lip
   290→      const cornerHighlight = Math.max(0, 1 - Math.abs(Math.abs(nx) - 0.7) * 3.2);
   291→      if (nz > 0) {
   292→        r += Math.round(cornerHighlight * 28 * KEY_LIGHT);
   293→        g += Math.round(cornerHighlight * 25 * KEY_LIGHT);
   294→        b += Math.round(cornerHighlight * 27 * KEY_LIGHT);
   295→      }
   296→
   297→      // Crimson rim light along outer side edges
   298→      if (Math.abs(nx) > 0.55) {
   299→        r += Math.round(24 * CRIMSON_RIM);
   300→        g += Math.round(4 * CRIMSON_RIM);
   301→        b += Math.round(10 * CRIMSON_RIM);
   302→      }
   303→
   304→      const segPath = pointsToSvgPath([
   305→        deckTopPts[i],
   306→        deckTopPts[next],
   307→        deckMidPts[next],
   308→        deckMidPts[i],
   309→      ]);
   310→      deckWallSegments.push({
   311→        d: segPath,
   312→        fill: `rgb(${Math.min(255, r)}, ${Math.min(255, g)}, ${Math.min(255, b)})`,
   313→      });
   314→    }
   315→
   316→    // Build tapered lower bevel segments between deckMidPts and deckPanPts
   317→    const lowerChamferPoly = pointsToSvgPath(deckMidPts);
   318→    const lowerPanPoly = pointsToSvgPath(deckPanPts);
   319→
   320→    // Front lip ergonomic CNC thumb notch (X in [-27, +27], Z = deckDepth, Y in [-4.4, 0])
   321→    const notchPts = [
   322→      projectPoint3D({ x: -27, y: 0, z: deckDepth }, cam),
   323→      projectPoint3D({ x: 27, y: 0, z: deckDepth }, cam),
   324→      projectPoint3D({ x: 24.5, y: -4.4, z: deckDepth }, cam),
   325→      projectPoint3D({ x: -24.5, y: -4.4, z: deckDepth }, cam),
   326→    ];
   327→    const notchBottomRim = [
   328→      projectPoint3D({ x: -24.5, y: -4.4, z: deckDepth }, cam),
   329→      projectPoint3D({ x: 24.5, y: -4.4, z: deckDepth }, cam),
   330→    ];
   331→
   332→    // Side ports on the wall that FACES the camera:
   333→    // with negative yaw the camera sits at +X, so the +halfW wall is visible.
   334→    const sideX = YAW_DEG <= 0 ? halfW : -halfW;
   335→    const makeSidePort = (zStart: number, zEnd: number, yTop: number, yBot: number) =>
   336→      pointsToSvgPath([
   337→        projectPoint3D({ x: sideX, y: yTop, z: zStart }, cam),
   338→        projectPoint3D({ x: sideX, y: yTop, z: zEnd }, cam),
   339→        projectPoint3D({ x: sideX, y: yBot, z: zEnd }, cam),
   340→        projectPoint3D({ x: sideX, y: yBot, z: zStart }, cam),
   341→      ]);
   342→
   343→    const sidePorts = [
   344→      // MagSafe 3 Port
   345→      { d: makeSidePort(28, 46, -2.8, -7.6), isMagSafe: true },
   346→      // Thunderbolt / USB-C Port 1
   347→      { d: makeSidePort(58, 71, -3.4, -7.2), isMagSafe: false },
   348→      // Thunderbolt / USB-C Port 2
   349→      { d: makeSidePort(82, 95, -3.4, -7.2), isMagSafe: false },
   350→      // 3.5mm Studio Audio Jack
   351→      { d: makeSidePort(110, 116, -3.8, -7.0), isMagSafe: false },
   352→    ];
   353→
   354→    // Four Rubber Feet underneath the chassis corners + their tight contact shadows
   355→    const footConfigs = [
   356→      { x: -halfW + 26, z: 26, r: 12 },
   357→      { x: halfW - 26, z: 26, r: 12 },
   358→      { x: -halfW + 26, z: deckDepth - 26, r: 12 },
   359→      { x: halfW - 26, z: deckDepth - 26, r: 12 },
   360→    ];
   361→
   362→    const rubberFeet = footConfigs.map((fc) => {
   363→      const topLoop: Point2D[] = [];
   364→      const botLoop: Point2D[] = [];
   365→      const shadowLoop: Point2D[] = [];
   366→      const steps = 16;
   367→      for (let i = 0; i < steps; i++) {
   368→        const a = (i / steps) * Math.PI * 2;
   369→        const px = fc.x + Math.cos(a) * fc.r;
   370→        const pz = fc.z + Math.sin(a) * fc.r;
   371→        topLoop.push(projectPoint3D({ x: px, y: basePanBottomY + 0.5, z: pz }, cam));
   372→        botLoop.push(projectPoint3D({ x: px, y: basePanBottomY - 2.2, z: pz }, cam));
   373→        shadowLoop.push(
   374→          projectPoint3D(
   375→            {
   376→              x: fc.x + Math.cos(a) * (fc.r * 1.45),
   377→              y: basePanBottomY - 2.5,
   378→              z: fc.z + Math.sin(a) * (fc.r * 1.45),
   379→            },
   380→            cam
   381→          )
   382→        );
   383→      }
   384→      return {
   385→        footPath: pointsToSvgPath(botLoop),
   386→        shadowPath: pointsToSvgPath(shadowLoop),
   387→      };
   388→    });
   389→
   390→    // Studio Contact Shadows on the ground plane (Y = basePanBottomY - 2.6)
   391→    // Kept tight and natural directly beneath the laptop chassis
   392→    const shadowOuterPts = sampleRoundedRect2D(
   393→      -halfW - 10,
   394→      halfW + 10,
   395→      -8,
   396→      deckDepth + 10,
   397→      [22, 22, 26, 26],
   398→      12
   399→    ).map((p) =>
   400→      projectPoint3D({ x: p.x, y: basePanBottomY - 2.8, z: p.y }, cam)
   401→    );
   402→
   403→    const shadowCorePts = sampleRoundedRect2D(
   404→      -halfW + 4,
   405→      halfW - 4,
   406→      4,
   407→      deckDepth - 2,
   408→      [14, 14, 16, 16],
   409→      12
   410→    ).map((p) =>
   411→      projectPoint3D({ x: p.x, y: basePanBottomY - 2.4, z: p.y }, cam)
   412→    );
   413→
   414→    // Upper display lid warm-white key light highlight path
   415→    const topKeyLightPts: Point2D[] = [];
   416→    for (let u = -halfW + 10; u <= halfW - 10; u += 12) {
   417→      topKeyLightPts.push(
   418→        projectPoint3D(lidPointToWorld(u, lidHeight - 0.4, -0.5), cam)
   419→      );
   420→    }
   421→
   422→    // Outer display lid crimson rim highlight paths
   423→    const rightCrimsonRimPts: Point2D[] = [];
   424→    for (let v = 10; v <= lidHeight - 10; v += 12) {
   425→      rightCrimsonRimPts.push(
   426→        projectPoint3D(lidPointToWorld(halfW - 0.3, v, -0.8), cam)
   427→      );
   428→    }
   429→
   430→    const leftCrimsonRimPts: Point2D[] = [];
   431→    for (let v = 10; v <= lidHeight - 10; v += 12) {
   432→      leftCrimsonRimPts.push(
   433→        projectPoint3D(lidPointToWorld(-halfW + 0.3, v, -0.8), cam)
   434→      );
   435→    }
   436→
   437→    return {
   438→      lidFrontPath: pointsToSvgPath(lidFrontPts),
   439→      lidBackPath: pointsToSvgPath(lidBackPts),
   440→      lidInnerPath: pointsToSvgPath(lidInnerPts),
   441→      lidRimSegments,
   442→      camCenter,
   443→      camSensorLeft,
   444→      hingePath,
   445→      deckTopPath: pointsToSvgPath(deckTopPts),
   446→      deckMidPath: lowerChamferPoly,
   447→      deckPanPath: lowerPanPoly,
   448→      deckWallSegments,
   449→      notchPath: pointsToSvgPath(notchPts),
   450→      notchBottomRim,
   451→      sidePorts,
   452→      rubberFeet,
   453→      shadowOuterPath: pointsToSvgPath(shadowOuterPts),
   454→      shadowCorePath: pointsToSvgPath(shadowCorePts),
   455→      topKeyLightPath: pointsToSvgPath(topKeyLightPts, false),
   456→      rightCrimsonRimPath: pointsToSvgPath(rightCrimsonRimPts, false),
   457→      leftCrimsonRimPath: pointsToSvgPath(leftCrimsonRimPts, false),
   458→    };
   459→  }, [
   460→    cam,
   461→    halfW,
   462→    lidHeight,
   463→    lidThickness,
   464→    deckDepth,
   465→    baseWallBottomY,
   466→    basePanBottomY,
   467→    lidPointToWorld,
   468→  ]);
   469→
   470→  return (
   471→    <div
   472→      style={{
   473→        width: `${STAGE_WIDTH}px`,
   474→        height: `${STAGE_HEIGHT}px`,
   475→        background: "transparent",
   476→      }}
   477→      className="relative select-none overflow-visible"
   478→    >
   479→      {/* LAYER 1: STUDIO CONTACT SHADOWS, LOWER CHASSIS BASE, AND 3D DISPLAY LID FRAME */}
   480→      <svg
   481→        width={STAGE_WIDTH}
   482→        height={STAGE_HEIGHT}
   483→        viewBox={`0 0 ${STAGE_WIDTH} ${STAGE_HEIGHT}`}
   484→        className="pointer-events-none absolute inset-0 overflow-visible"
   485→      >
   486→        <defs>
   487→          {/* Tight Natural Studio Contact Shadow Blur */}
   488→          <filter
   489→            id="contact-shadow-outer"
   490→            x="-20%"
   491→            y="-20%"
   492→            width="140%"
   493→            height="140%"
   494→          >
   495→            <feGaussianBlur stdDeviation="22" />
   496→          </filter>
   497→          <filter
   498→            id="contact-shadow-core"
   499→            x="-15%"
   500→            y="-15%"
   501→            width="130%"
   502→            height="130%"
   503→          >
   504→            <feGaussianBlur stdDeviation="9" />
   505→          </filter>
   506→          <filter
   507→            id="foot-shadow-blur"
   508→            x="-30%"
   509→            y="-30%"
   510→            width="160%"
   511→            height="160%"
   512→          >
   513→            <feGaussianBlur stdDeviation="3.5" />
