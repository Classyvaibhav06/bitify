    41→    41→
    42→    42→export const STAGE_WIDTH = 1440;
    43→    43→export const STAGE_HEIGHT = 960;
    44→    44→
    45→    45→interface StudioLaptopRenderProps {
    46→    46→  settings: StudioRenderSettings;
    47→    47→  stageRef?: React.RefObject<HTMLDivElement | null>;
    48→    48→}
    49→    49→
    50→    50→export const StudioLaptopRender: React.FC<StudioLaptopRenderProps> = ({
    51→    51→  settings,
    52→    52→  stageRef,
    53→    53→}) => {
    54→    54→  const cam: CameraConfig = useMemo(
    55→    55→    () => ({
    56→    56→      yawDeg: settings.yawDeg,
    57→    57→      pitchDeg: settings.pitchDeg,
    58→    58→      lidAngleDeg: settings.lidAngleDeg,
    59→    59→      distance: 1060,
    60→    60→      focalLength: 2280,
    61→    61→      target: { x: -8, y: 60, z: 98 },
    62→    62→      viewportCenter: { x: STAGE_WIDTH * 0.5, y: STAGE_HEIGHT * 0.5 },
    63→    63→    }),
    64→    64→    [settings.yawDeg, settings.pitchDeg, settings.lidAngleDeg]
    65→    65→  );
    66→    66→
    67→    67→  // Physical dimensions in millimeter-scale 3D space
    68→    68→  const halfW = 178; // 356mm total width
    69→    69→  const deckDepth = 246; // 246mm deck depth
    70→    70→  const lidHeight = 234; // 234mm display lid height
    71→    71→  const lidThickness = 5.0; // Slim 5.0mm anodized aluminum lid
    72→    72→  const baseWallBottomY = -10.8; // Vertical side wall bottom
    73→    73→  const basePanBottomY = -13.8; // Tapered lower chassis pan
    74→    74→
    75→    75→  // Convert display lid local coordinates (u, v, w) to 3D world space
    76→    76→  // u: [-halfW..+halfW], v: [0..lidHeight] along lid face, w: [-lidThickness..0] depth into back shell
    77→    77→  const lidPointToWorld = useMemo(() => {
    78→    78→    const tiltBackRad = ((settings.lidAngleDeg - 90) * Math.PI) / 180;
    79→    79→    const cosT = Math.cos(tiltBackRad);
    80→    80→    const sinT = Math.sin(tiltBackRad);
    81→    81→    const hingeY = 1.2;
    82→    82→    const hingeZ = 7.5;
    83→    83→
    84→    84→    return (u: number, v: number, w = 0): Vec3 => ({
    85→    85→      x: u,
    86→    86→      y: hingeY + v * cosT - w * sinT,
    87→    87→      z: hingeZ - v * sinT + w * cosT,
    88→    88→    });
    89→    89→  }, [settings.lidAngleDeg]);
    90→    90→
    91→    91→  // 1. Compute 4 Projected Corners of the Active Display Screen Aperture
    92→    92→  // Physical screen aperture: u in [-172.5, +172.5] (345mm), v in [11.2, 226.8] (215.6mm) -> exact 16:10
    93→    93→  const screenQuad = useMemo((): [Point2D, Point2D, Point2D, Point2D] => {
    94→    94→    const uMin = -172.4;
    95→    95→    const uMax = 172.4;
    96→    96→    const vBottom = 11.2;
    97→    97→    const vTop = 226.8;
    98→    98→
    99→    99→    const tl = projectPoint3D(lidPointToWorld(uMin, vTop, 0), cam);
   100→   100→    const tr = projectPoint3D(lidPointToWorld(uMax, vTop, 0), cam);
   101→   101→    const br = projectPoint3D(lidPointToWorld(uMax, vBottom, 0), cam);
   102→   102→    const bl = projectPoint3D(lidPointToWorld(uMin, vBottom, 0), cam);
   103→   103→    return [tl, tr, br, bl];
   104→   104→  }, [cam, lidPointToWorld]);
   105→   105→
   106→   106→  const screenMatrix3d = useMemo(
   107→   107→    () =>
   108→   108→      computeHomographyMatrix3d(
   109→   109→        SCREEN_PIXEL_WIDTH,
   110→   110→        SCREEN_PIXEL_HEIGHT,
   111→   111→        screenQuad
   112→   112→      ),
   113→   113→    [screenQuad]
   114→   114→  );
   115→   115→
   116→   116→  // 2. Compute 4 Projected Corners of the Top Keyboard Deck Plane (Y = 0)
   117→   117→  const deckQuad = useMemo((): [Point2D, Point2D, Point2D, Point2D] => {
   118→   118→    const tl = projectPoint3D({ x: -halfW, y: 0, z: 0 }, cam);
   119→   119→    const tr = projectPoint3D({ x: halfW, y: 0, z: 0 }, cam);
   120→   120→    const br = projectPoint3D({ x: halfW, y: 0, z: deckDepth }, cam);
   121→   121→    const bl = projectPoint3D({ x: -halfW, y: 0, z: deckDepth }, cam);
   122→   122→    return [tl, tr, br, bl];
   123→   123→  }, [cam, halfW, deckDepth]);
   124→   124→
   125→   125→  const deckMatrix3d = useMemo(
   126→   126→    () =>
   127→   127→      computeHomographyMatrix3d(DECK_PIXEL_WIDTH, DECK_PIXEL_HEIGHT, deckQuad),
   128→   128→    [deckQuad]
   129→   129→  );
   130→   130→
   131→   131→  // 3. Sample 3D Rounded Contours for Lid, Screen Bezel, Deck, Lower Chassis, and Shadows
   132→   132→  const geometryPaths = useMemo(() => {
   133→   133→    // Lid 2D local rounded contour (note: in 2D sampling yMin is bottom v=0, yMax is top v=lidHeight)
   134→   134→    const lidContourLocal = sampleRoundedRect2D(
   135→   135→      -halfW,
   136→   136→      halfW,
   137→   137→      0,
   138→   138→      lidHeight,
   139→   139→      [5, 5, 11.5, 11.5],
   140→   140→      12
   141→   141→    );
   142→   142→
   143→   143→    const lidFrontPts = lidContourLocal.map((p) =>
   144→   144→      projectPoint3D(lidPointToWorld(p.x, p.y, 0), cam)
   145→   145→    );
   146→   146→    const lidBackPts = lidContourLocal.map((p) =>
   147→   147→      projectPoint3D(lidPointToWorld(p.x, p.y, -lidThickness), cam)
   148→   148→    );
   149→   149→
   150→   150→    // Lid inner bezel step (1.5mm inset)
   151→   151→    const lidInnerPts = sampleRoundedRect2D(
   152→   152→      -halfW + 1.6,
   153→   153→      halfW - 1.6,
   154→   154→      1.4,
   155→   155→      lidHeight - 1.6,
   156→   156→      [4, 4, 10, 10],
   157→   157→      10
   158→   158→    ).map((p) => projectPoint3D(lidPointToWorld(p.x, p.y, 0), cam));
   159→   159→
   160→   160→    // Build 3D extruded rim quads between lidBackPts and lidFrontPts
   161→   161→    const lidRimSegments: {
   162→   162→      d: string;
   163→   163→      color: string;
   164→   164→      isTopKeyLight: boolean;
   165→   165→      isCrimsonRim: boolean;
   166→   166→    }[] = [];
   167→   167→
   168→   168→    for (let i = 0; i < lidFrontPts.length; i++) {
   169→   169→      const next = (i + 1) % lidFrontPts.length;
   170→   170→      const pL1 = lidContourLocal[i];
   171→   171→      const pL2 = lidContourLocal[next];
   172→   172→      const midU = (pL1.x + pL2.x) * 0.5;
   173→   173→      const midV = (pL1.y + pL2.y) * 0.5;
   174→   174→
   175→   175→      const isTop = midV > lidHeight - 14;
   176→   176→      const isRightSide = midU > halfW - 14;
   177→   177→      const isLeftSide = midU < -halfW + 14;
   178→   178→
   179→   179→      let fill = '#16161B';
   180→   180→      if (isTop) {
   181→   181→        fill = '#383436'; // Warm-white key light grazing upper edge
   182→   182→      } else if (isRightSide || isLeftSide) {
   183→   183→        fill = '#2A1720'; // Subtle crimson rim light along outer edges
   184→   184→      }
   185→   185→
   186→   186→      const segPath = pointsToSvgPath(
   187→   187→        [lidBackPts[i], lidBackPts[next], lidFrontPts[next], lidFrontPts[i]],
   188→   188→        true
   189→   189→      );
   190→   190→      lidRimSegments.push({
   191→   191→        d: segPath,
   192→   192→        color: fill,
   193→   193→        isTopKeyLight: isTop,
   194→   194→        isCrimsonRim: isRightSide || isLeftSide,
   195→   195→      });
   196→   196→    }
   197→   197→
   198→   198→    // Top camera notch/sensor coordinates on upper bezel
   199→   199→    const camCenter = projectPoint3D(
   200→   200→      lidPointToWorld(0, lidHeight - 3.8, 0),
   201→   201→      cam
   202→   202→    );
   203→   203→    const camSensorLeft = projectPoint3D(
   204→   204→      lidPointToWorld(-4.5, lidHeight - 3.8, 0),
   205→   205→      cam
   206→   206→    );
   207→   207→
   208→   208→    // Cylindrical hinge barrel between (X in [-132, +132])
   209→   209→    const hingeTopLeft = projectPoint3D({ x: -134, y: 4.2, z: 6.5 }, cam);
   210→   210→    const hingeTopRight = projectPoint3D({ x: 134, y: 4.2, z: 6.5 }, cam);
   211→   211→    const hingeBotRight = projectPoint3D({ x: 134, y: -1.2, z: 11.5 }, cam);
   212→   212→    const hingeBotLeft = projectPoint3D({ x: -134, y: -1.2, z: 11.5 }, cam);
   213→   213→    const hingePath = pointsToSvgPath([
   214→   214→      hingeTopLeft,
   215→   215→      hingeTopRight,
   216→   216→      hingeBotRight,
   217→   217→      hingeBotLeft,
   218→   218→    ]);
   219→   219→
   220→   220→    // Deck 2D local contour: X in [-halfW..+halfW], Z in [0..deckDepth]
   221→   221→    const deckContourLocal = sampleRoundedRect2D(
   222→   222→      -halfW,
   223→   223→      halfW,
   224→   224→      0,
   225→   225→      deckDepth,
   226→   226→      [11, 11, 13.5, 13.5],
   227→   227→      14
   228→   228→    );
   229→   229→
   230→   230→    const deckTopPts = deckContourLocal.map((p) =>
   231→   231→      projectPoint3D({ x: p.x, y: 0, z: p.y }, cam)
   232→   232→    );
   233→   233→    const deckMidPts = deckContourLocal.map((p) =>
   234→   234→      projectPoint3D({ x: p.x, y: baseWallBottomY, z: p.y }, cam)
   235→   235→    );
   236→   236→
   237→   237→    // Inset tapered lower pan contour
   238→   238→    const panContourLocal = sampleRoundedRect2D(
   239→   239→      -halfW + 5.5,
   240→   240→      halfW - 5.5,
   241→   241→      5.5,
   242→   242→      deckDepth - 5.5,
   243→   243→      [10, 10, 12, 12],
   244→   244→      12
   245→   245→    );
   246→   246→    const deckPanPts = panContourLocal.map((p) =>
   247→   247→      projectPoint3D({ x: p.x, y: basePanBottomY, z: p.y }, cam)
   248→   248→    );
   249→   249→
   250→   250→    // Build 3D vertical wall segments around the unibody deck perimeter
   251→   251→    const deckWallSegments: { d: string; fill: string }[] = [];
   252→   252→    for (let i = 0; i < deckTopPts.length; i++) {
   253→   253→      const next = (i + 1) % deckTopPts.length;
   254→   254→      const c1 = deckContourLocal[i];
   255→   255→      const c2 = deckContourLocal[next];
   256→   256→
   257→   257→      // Outward 2D normal in (X, Z) plane
   258→   258→      const dx = c2.x - c1.x;
   259→   259→      const dz = c2.y - c1.y;
   260→   260→      const len = Math.hypot(dx, dz) || 1;
   261→   261→      const nx = dz / len;
   262→   262→      const nz = -dx / len;
   263→   263→
   264→   264→      // Shade based on normal orientation (front lip vs left/right side walls vs corner curve)
   265→   265→      let r = 20;
   266→   266→      let g = 20;
   267→   267→      let b = 25;
   268→   268→
   269→   269→      // Front-facing lip receives soft warm studio fill
   270→   270→      if (nz > 0.4) {
   271→   271→        const frontFactor = (nz - 0.4) / 0.6;
   272→   272→        r += Math.round(frontFactor * 14 * settings.keyLightIntensity);
   273→   273→        g += Math.round(frontFactor * 13 * settings.keyLightIntensity);
   274→   274→        b += Math.round(frontFactor * 15 * settings.keyLightIntensity);
   275→   275→      }
   276→   276→
   277→   277→      // Corner specular catchlight where side wall curves into front lip
   278→   278→      const cornerHighlight = Math.max(0, 1 - Math.abs(Math.abs(nx) - 0.7) * 3.2);
   279→   279→      if (nz > 0) {
   280→   280→        r += Math.round(cornerHighlight * 28 * settings.keyLightIntensity);
   281→   281→        g += Math.round(cornerHighlight * 25 * settings.keyLightIntensity);
   282→   282→        b += Math.round(cornerHighlight * 27 * settings.keyLightIntensity);
   283→   283→      }
   284→   284→
   285→   285→      // Crimson rim light along outer side edges
   286→   286→      if (Math.abs(nx) > 0.55) {
   287→   287→        r += Math.round(24 * settings.crimsonRimIntensity);
   288→   288→        g += Math.round(4 * settings.crimsonRimIntensity);
   289→   289→        b += Math.round(10 * settings.crimsonRimIntensity);
   290→   290→      }
   291→   291→
   292→   292→      const segPath = pointsToSvgPath([
   293→   293→        deckTopPts[i],
   294→   294→        deckTopPts[next],
   295→   295→        deckMidPts[next],
   296→   296→        deckMidPts[i],
   297→   297→      ]);
   298→   298→      deckWallSegments.push({
   299→   299→        d: segPath,
   300→   300→        fill: `rgb(${Math.min(255, r)}, ${Math.min(255, g)}, ${Math.min(255, b)})`,
   301→   301→      });
   302→   302→    }
   303→   303→
   304→   304→    // Build tapered lower bevel segments between deckMidPts and deckPanPts
   305→   305→    const lowerChamferPoly = pointsToSvgPath(deckMidPts);
   306→   306→    const lowerPanPoly = pointsToSvgPath(deckPanPts);
   307→   307→
   308→   308→    // Front lip ergonomic CNC thumb notch (X in [-27, +27], Z = deckDepth, Y in [-4.4, 0])
   309→   309→    const notchPts = [
   310→   310→      projectPoint3D({ x: -27, y: 0, z: deckDepth }, cam),
   311→   311→      projectPoint3D({ x: 27, y: 0, z: deckDepth }, cam),
   312→   312→      projectPoint3D({ x: 24.5, y: -4.4, z: deckDepth }, cam),
   313→   313→      projectPoint3D({ x: -24.5, y: -4.4, z: deckDepth }, cam),
   314→   314→    ];
   315→   315→    const notchBottomRim = [
   316→   316→      projectPoint3D({ x: -24.5, y: -4.4, z: deckDepth }, cam),
   317→   317→      projectPoint3D({ x: 24.5, y: -4.4, z: deckDepth }, cam),
   318→   318→    ];
   319→   319→
   320→   320→    // Side ports on whichever side wall faces the camera (left wall X = -halfW when yaw < 0, right wall X = +halfW when yaw > 0)
   321→   321→    const sideX = settings.yawDeg <= 0 ? -halfW : halfW;
   322→   322→    const makeSidePort = (zStart: number, zEnd: number, yTop: number, yBot: number) =>
   323→   323→      pointsToSvgPath([
   324→   324→        projectPoint3D({ x: sideX, y: yTop, z: zStart }, cam),
   325→   325→        projectPoint3D({ x: sideX, y: yTop, z: zEnd }, cam),
   326→   326→        projectPoint3D({ x: sideX, y: yBot, z: zEnd }, cam),
   327→   327→        projectPoint3D({ x: sideX, y: yBot, z: zStart }, cam),
   328→   328→      ]);
   329→   329→
   330→   330→    const sidePorts = [
   331→   331→      // MagSafe 3 Port
   332→   332→      { d: makeSidePort(28, 46, -2.8, -7.6), isMagSafe: true },
   333→   333→      // Thunderbolt / USB-C Port 1
   334→   334→      { d: makeSidePort(58, 71, -3.4, -7.2), isMagSafe: false },
   335→   335→      // Thunderbolt / USB-C Port 2
   336→   336→      { d: makeSidePort(82, 95, -3.4, -7.2), isMagSafe: false },
   337→   337→      // 3.5mm Studio Audio Jack
   338→   338→      { d: makeSidePort(110, 116, -3.8, -7.0), isMagSafe: false },
   339→   339→    ];
   340→   340→
   341→   341→    // Four Rubber Feet underneath the chassis corners + their tight contact shadows
   342→   342→    const footConfigs = [
   343→   343→      { x: -halfW + 26, z: 26, r: 12 },
   344→   344→      { x: halfW - 26, z: 26, r: 12 },
   345→   345→      { x: -halfW + 26, z: deckDepth - 26, r: 12 },
   346→   346→      { x: halfW - 26, z: deckDepth - 26, r: 12 },
   347→   347→    ];
   348→   348→
   349→   349→    const rubberFeet = footConfigs.map((fc) => {
   350→   350→      const topLoop: Point2D[] = [];
   351→   351→      const botLoop: Point2D[] = [];
   352→   352→      const shadowLoop: Point2D[] = [];
   353→   353→      const steps = 16;
   354→   354→      for (let i = 0; i < steps; i++) {
   355→   355→        const a = (i / steps) * Math.PI * 2;
   356→   356→        const px = fc.x + Math.cos(a) * fc.r;
   357→   357→        const pz = fc.z + Math.sin(a) * fc.r;
   358→   358→        topLoop.push(projectPoint3D({ x: px, y: basePanBottomY + 0.5, z: pz }, cam));
   359→   359→        botLoop.push(projectPoint3D({ x: px, y: basePanBottomY - 2.2, z: pz }, cam));
   360→   360→        shadowLoop.push(
   361→   361→          projectPoint3D(
   362→   362→            {
   363→   363→              x: fc.x + Math.cos(a) * (fc.r * 1.45),
   364→   364→              y: basePanBottomY - 2.5,
   365→   365→              z: fc.z + Math.sin(a) * (fc.r * 1.45),
   366→   366→            },
   367→   367→            cam
   368→   368→          )
   369→   369→        );
   370→   370→      }
   371→   371→      return {
   372→   372→        footPath: pointsToSvgPath(botLoop),
   373→   373→        shadowPath: pointsToSvgPath(shadowLoop),
   374→   374→      };
   375→   375→    });
   376→   376→
   377→   377→    // Studio Contact Shadows on the ground plane (Y = basePanBottomY - 2.6)
   378→   378→    // Kept tight and natural directly beneath the laptop chassis
   379→   379→    const shadowOuterPts = sampleRoundedRect2D(
   380→   380→      -halfW - 10,
   381→   381→      halfW + 10,
   382→   382→      -8,
   383→   383→      deckDepth + 10,
   384→   384→      [22, 22, 26, 26],
   385→   385→      12
   386→   386→    ).map((p) =>
   387→   387→      projectPoint3D({ x: p.x, y: basePanBottomY - 2.8, z: p.y }, cam)
   388→   388→    );
   389→   389→
   390→   390→    const shadowCorePts = sampleRoundedRect2D(
   391→   391→      -halfW + 4,
   392→   392→      halfW - 4,
   393→   393→      4,
   394→   394→      deckDepth - 2,
   395→   395→      [14, 14, 16, 16],
   396→   396→      12
   397→   397→    ).map((p) =>
   398→   398→      projectPoint3D({ x: p.x, y: basePanBottomY - 2.4, z: p.y }, cam)
   399→   399→    );
   400→   400→
   401→   401→    // Upper display lid warm-white key light highlight path
   402→   402→    const topKeyLightPts: Point2D[] = [];
   403→   403→    for (let u = -halfW + 10; u <= halfW - 10; u += 12) {
   404→   404→      topKeyLightPts.push(
   405→   405→        projectPoint3D(lidPointToWorld(u, lidHeight - 0.4, -0.5), cam)
   406→   406→      );
   407→   407→    }
   408→   408→
   409→   409→    // Outer display lid crimson rim highlight path
   410→   410→    const rightCrimsonRimPts: Point2D[] = [];
   411→   411→    for (let v = 10; v <= lidHeight - 10; v += 12) {
   412→   412→      rightCrimsonRimPts.push(
   413→   413→        projectPoint3D(lidPointToWorld(halfW - 0.3, v, -0.8), cam)
   414→   414→      );
   415→   415→    }
   416→   416→
   417→   417→    const leftCrimsonRimPts: Point2D[] = [];
   418→   418→    for (let v = 10; v <= lidHeight - 10; v += 12) {
   419→   419→      leftCrimsonRimPts.push(
   420→   420→        projectPoint3D(lidPointToWorld(-halfW + 0.3, v, -0.8), cam)
   421→   421→      );
   422→   422→    }
   423→   423→
   424→   424→    return {
   425→   425→      lidFrontPath: pointsToSvgPath(lidFrontPts),
   426→   426→      lidBackPath: pointsToSvgPath(lidBackPts),
