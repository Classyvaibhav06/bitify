    20→    20→ * Studio laptop render — ported from the Gemini-authored Bitify laptop
    21→    21→ * render, adapted for the Bitify hero:
    22→    22→ *
    23→    23→ *  - The screen now carries the REAL Bitify desktop screenshot
    24→    24→ *    (`desktop-hero.webp`, 1403 x 930, ratio 1.5086). The physical
    25→    25→ *    screen aperture and the lid height were re-derived so the

    33→    33→ *
    34→    34→ * The laptop is built from a real pinhole-camera projection: every
    35→    35→ * chassis edge, wall, hinge and shadow is a 3D point projected to 2D,
    36→    36→ * and the screen + keyboard deck are flat surfaces mapped on with
    37→    37→ * solved homography matrix3d transforms. The default pose faces the
    38→    38→ * viewer with a slight left bias — perfect for a right-hand hero
    39→    39→ * placement looking back at the headline.
    40→    40→ */
    41→    41→

    43→    43→ * the image maps 1:1 onto the projected screen quad. */

    55→    55→const LID_ANGLE_DEG = 102.5;

    77→    77→  // Physical dimensions in millimeter-scale 3D space

    83→    83→  const lidThickness = 5.0; // Slim 5.0mm anodized aluminum lid

    87→    87→  // Convert display lid local coordinates (u, v, w) to 3D world space

    90→    90→    const tiltBackRad = ((LID_ANGLE_DEG - 90) * Math.PI) / 180;

    98→    98→      y: hingeY + v * cosT - w * sinT,

   100→   100→    });
   101→   101→  }, []);
   102→   102→

   156→   156→      projectPoint3D(lidPointToWorld(p.x, p.y, 0), cam)
   157→   157→    );

   159→   159→      projectPoint3D(lidPointToWorld(p.x, p.y, -lidThickness), cam)

   183→   183→      const pL2 = lidContourLocal[next];

   235→   235→      halfW,
   236→   236→      0,
   237→   237→      deckDepth,

   241→   241→
   242→   242→    const deckTopPts = deckContourLocal.map((p) =>

   246→   246→      projectPoint3D({ x: p.x, y: baseWallBottomY, z: p.y }, cam)
   247→   247→    );
   248→   248→
   249→   249→    // Inset tapered lower pan contour
   250→   250→    const panContourLocal = sampleRoundedRect2D(
   251→   251→      -halfW + 5.5,
   252→   252→      halfW - 5.5,
   253→   253→      5.5,
   254→   254→      deckDepth - 5.5,
   255→   255→      [10, 10, 12, 12],
   256→   256→      12

   258→   258→    const deckPanPts = panContourLocal.map((p) =>
   259→   259→      projectPoint3D({ x: p.x, y: basePanBottomY, z: p.y }, cam)

   261→   261→
   262→   262→    // Build 3D vertical wall segments around the unibody deck perimeter
   263→   263→    const deckWallSegments: { d: string; fill: string }[] = [];
   264→   264→    for (let i = 0; i < deckTopPts.length; i++) {
   265→   265→      const next = (i + 1) % deckTopPts.length;
   266→   266→      const c1 = deckContourLocal[i];
   267→   267→      const c2 = deckContourLocal[next];

   269→   269→      // Outward 2D normal in (X, Z) plane

   271→   271→      const dz = c2.y - c1.y;

   292→   292→        r += Math.round(cornerHighlight * 28 * KEY_LIGHT);

   306→   306→        deckTopPts[next],
   307→   307→        deckMidPts[next],
   308→   308→        deckMidPts[i],
   309→   309→      ]);

   311→   311→        d: segPath,
   312→   312→        fill: `rgb(${Math.min(255, r)}, ${Math.min(255, g)}, ${Math.min(255, b)})`,
   313→   313→      });
   314→   314→    }
   315→   315→
